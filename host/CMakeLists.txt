cmake_minimum_required(VERSION 3.16)
project(smart_scale_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE STRING "LVGL config" FORCE)
set(LV_CONF_INCLUDE_SIMPLE ON CACHE BOOL "" FORCE)
add_subdirectory(../.tools/lvgl lvgl EXCLUDE_FROM_ALL)
add_executable(smart_scale_host main.cxx ../main/ui.cxx ../main/weight_font_72.c ../main/session.cxx)
target_include_directories(smart_scale_host PRIVATE ../main)
target_compile_options(smart_scale_host PRIVATE -Wall -Wextra -Werror)
target_link_libraries(smart_scale_host PRIVATE lvgl m)
find_package(SDL2 QUIET)
if(SDL2_FOUND)
    target_compile_definitions(smart_scale_host PRIVATE SMART_SCALE_SDL=1)
    target_link_libraries(smart_scale_host PRIVATE SDL2::SDL2)
endif()
